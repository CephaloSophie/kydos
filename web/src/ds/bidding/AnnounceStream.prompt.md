```jsx
<AnnounceStream entries={[
  {player:'Vous', team:'a', kind:'bid', value:90, suit:'spades'},
  {player:'Iznogoud', team:'b', isRobot:true, kind:'contre'},
  {player:'Partenaire', team:'a', kind:'surcontre'},
]} />
```
